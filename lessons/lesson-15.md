## 2. Главная учебная цель и проверяемый результат

- **Главная учебная цель:** Продемонстрировать целостное системное мышление: объединить микроконтроллер Arduino (сбор данных с датчиков и управление аварийным маяком) и компьютер на Linux (приём телеметрии по Serial/сети, анализ рисков и запись в бортовой журнал); самостоятельно диагностировать и исправить скрытую программную ошибку.
- **Проверяемый результат (действие ребёнка без подсказки):**
  1. Самостоятельно запускает комплекс: прошивает Arduino, запускает в терминале программу-демон на C `telemetry_logger`, проверяет стабильность потока данных.

  3. В ответ на неожиданную задачу преподавателя («А сделай так, чтобы при тревоге маяк мигал с двойной частотой, а в лог писался код угрозы RED_ALERT») за 7–10 минут придумывает план, вносит изменения в код, компилирует и показывает работающий результат.

## 4. Исходные коды проекта «Кибер-Маяк»

### 1. Прошивка микроконтроллера Arduino (`beacon_firmware.ino`)
```cpp
const int PIN_BUMPER    = 2;  // Кнопка-бампер (вход с подтяжкой)
const int PIN_LIGHT     = A0; // Фоторезистор (вход АЦП)
const int PIN_ALARM_RED = 12; // Красный стробоскоп (цифровой выход)
const int PIN_STATUS_BL = 9;  // Синий дежурный маяк (ШИМ-выход)

int light_threshold = 300; // Порог темноты/тени

void setup() {
    Serial.begin(9600);
    pinMode(PIN_BUMPER, INPUT_PULLUP);
    pinMode(PIN_ALARM_RED, OUTPUT);
    pinMode(PIN_STATUS_BL, OUTPUT);

    // Приветственный сигнал запуска
    digitalWrite(PIN_ALARM_RED, HIGH);
    delay(200);
    digitalWrite(PIN_ALARM_RED, LOW);
    Serial.println("[SYSTEM_INIT] Станция Кибер-Маяк запущена.");
}

void loop() {
    int light_val = analogRead(PIN_LIGHT);
    int bumper_val = digitalRead(PIN_BUMPER); // LOW = нажат!
    int is_alarm = 0;

    // Условие тревоги: либо перекрыт свет (тень нарушителя), либо нажат бампер взлома!
    if (bumper_val == LOW || light_val < light_threshold) {
        is_alarm = 1;
        digitalWrite(PIN_ALARM_RED, HIGH);
        analogWrite(PIN_STATUS_BL, 0); // гасим дежурный свет
    } else {
        is_alarm = 0;
        digitalWrite(PIN_ALARM_RED, LOW);
        // Дежурный режим: синий свет плавно горит в четверть накала
        analogWrite(PIN_STATUS_BL, 64);
    }

    // Отправка пакета телеметрии в последовательный порт:
    Serial.print("DATA light=");
    Serial.print(light_val);
    Serial.print(" bumper=");
    Serial.print(bumper_val == LOW ? "HIT" : "OK");
    Serial.print(" alarm=");
    Serial.println(is_alarm);

    delay(250); // Частота телеметрии 4 раза в секунду
}
```

### 2. Серверная программа-монитор на C в Linux (`telemetry_daemon.c`)
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Функция анализа пакета телеметрии
void process_packet(int packet_id, int light, int is_alarm) {
    printf("[ПАКЕТ #%04d] Освещённость: %4d | ", packet_id, light);

    if (is_alarm) {
        printf(">>> [ТРЕВОГА! ПЕРИМЕТР НАРУШЕН!] <<<\n");
    } else {
        printf("Периметр стабилен. Станция в дежурном режиме.\n");
    }
}

int main() {
    printf("==============================================\n");
    printf("  ЦЕНТРАЛЬНЫЙ ДИСПЕТЧЕРСКИЙ ПУЛЬТ БАЗЫ (LINUX)\n");
    printf("  ПРОТОКОЛ ПРИЁМА ТЕЛЕМЕТРИИ С КИБЕР-МАЯКА     \n");
    printf("==============================================\n");

    int packet_counter = 0;
    int test_light_stream[] = {520, 515, 490, 210, 180, 510};
    int test_alarm_stream[] = {  0,   0,   0,   1,   1,   0};
    int total_samples = 6;

    for (int i = 0; i < total_samples; i++) {
        packet_counter++;
        process_packet(packet_counter, test_light_stream[i], test_alarm_stream[i]);
    }

    printf("==============================================\n");
    printf("Сеанс связи завершён успешно. Потерь пакетов нет.\n");
    return 0;
}
```
